export type SpeechFailure = "unsupported" | "offline" | "permission" | "microphone" | "language" | "silence" | "cancelled";
export class SpeechError extends Error {
  reason: SpeechFailure;
  constructor(reason: SpeechFailure) { super(reason); this.reason = reason; }
}
export type SpeechResult = { text: string; confidence: number; speechStartedAt: number | null };
export interface SpeechProvider {
  speak(text: string, language: string, signal?: AbortSignal): Promise<number>;
  listen(language: string, signal?: AbortSignal): Promise<SpeechResult>;
  cancel(): void;
}

interface Recognition {
  lang: string; interimResults: boolean; continuous: boolean;
  onresult: ((event: { results: { [n: number]: { [n: number]: { transcript: string; confidence: number } } } }) => void) | null;
  onerror: ((event: { error: string }) => void) | null;
  onend: (() => void) | null; onspeechstart: (() => void) | null;
  start(): void; abort(): void;
}
type SpeechWindow = Window & { SpeechRecognition?: new () => Recognition; webkitSpeechRecognition?: new () => Recognition };

export function recognitionFailure(code: string): SpeechFailure {
  return ({ "not-allowed": "permission", "service-not-allowed": "permission", "audio-capture": "microphone", "language-not-supported": "language", network: "offline", "no-speech": "silence", aborted: "cancelled" } as Record<string, SpeechFailure>)[code] ?? "unsupported";
}

// Web Speech uses one browser-wide audio channel. Only its owner may cancel it.
let owner: BrowserSpeechProvider | null = null;
export class BrowserSpeechProvider implements SpeechProvider {
  private cancelPending: (() => void) | null = null;
  private generation = 0;
  cancel() {
    this.generation++;
    this.cancelPending?.();
    this.cancelPending = null;
    if (owner === this) owner = null;
  }
  private acquire() {
    this.cancel();
    owner?.cancel();
    owner = this;
    return this.generation;
  }
  private release() {
    this.cancelPending = null;
    if (owner === this) owner = null;
  }
  async speak(text: string, language: string, signal?: AbortSignal): Promise<number> {
    if (signal?.aborted) throw new SpeechError("cancelled");
    if (typeof window === "undefined" || !("speechSynthesis" in window)) throw new SpeechError("unsupported");
    const generation = this.acquire();
    const synthesis = window.speechSynthesis;
    try {
      // Voice discovery is cancellable too; navigation must not start stale audio later.
      if (!synthesis.getVoices().length) await new Promise<void>((resolve, reject) => {
        const cleanup = () => { clearTimeout(timer); synthesis.removeEventListener("voiceschanged", finish); signal?.removeEventListener("abort", abort); };
        const finish = () => { cleanup(); resolve(); };
        const abort = () => { cleanup(); reject(new SpeechError("cancelled")); };
        const timer = setTimeout(finish, 700);
        this.cancelPending = abort;
        signal?.addEventListener("abort", abort, { once: true });
        synthesis.addEventListener("voiceschanged", finish);
      });
      if (signal?.aborted || generation !== this.generation) throw new SpeechError("cancelled");
      const voice = synthesis.getVoices().find(v => v.lang.toLowerCase().split("-")[0] === language.toLowerCase().split("-")[0]);
      if (!voice) throw new SpeechError("language");
      return await new Promise<number>((resolve, reject) => {
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = language; utterance.voice = voice; utterance.rate = 0.85;
        let settled = false;
        const finish = (failure?: SpeechFailure) => {
          if (settled) return; settled = true;
          clearTimeout(timer); signal?.removeEventListener("abort", abort);
          utterance.onend = null; utterance.onerror = null;
          if (failure) { try { synthesis.cancel(); } catch {} reject(new SpeechError(failure)); }
          else resolve(performance.now());
        };
        const abort = () => finish("cancelled");
        // Watchdog for missing browser callbacks, never a patient answer deadline.
        const timer = setTimeout(() => finish("unsupported"), Math.max(30000, text.length * 150 + 10000));
        this.cancelPending = abort;
        signal?.addEventListener("abort", abort, { once: true });
        utterance.onend = () => finish();
        utterance.onerror = () => finish("unsupported");
        try { synthesis.speak(utterance); } catch { finish("unsupported"); }
      });
    } finally {
      if (generation === this.generation) this.release();
    }
  }
  listen(language: string, signal?: AbortSignal): Promise<SpeechResult> {
    if (signal?.aborted) return Promise.reject(new SpeechError("cancelled"));
    if (typeof navigator === "undefined" || !navigator.onLine) return Promise.reject(new SpeechError("offline"));
    const SpeechRecognition = (window as SpeechWindow).SpeechRecognition ?? (window as SpeechWindow).webkitSpeechRecognition;
    if (!SpeechRecognition) return Promise.reject(new SpeechError("unsupported"));
    const generation = this.acquire();
    return new Promise<SpeechResult>((resolve, reject) => {
      const recognition = new SpeechRecognition();
      recognition.lang = language; recognition.interimResults = false; recognition.continuous = false;
      let started: number | null = null, settled = false;
      const finish = (result?: SpeechResult, failure: SpeechFailure = "silence") => {
        if (settled) return; settled = true;
        clearTimeout(timer); signal?.removeEventListener("abort", abort);
        recognition.onend = null; recognition.onresult = null; recognition.onerror = null; recognition.onspeechstart = null;
        try { recognition.abort(); } catch { /* Some engines throw after recognition has already stopped. */ }
        if (result) resolve(result); else reject(new SpeechError(failure));
      };
      const abort = () => finish(undefined, "cancelled");
      const timer = setTimeout(() => finish(), 15000);
      this.cancelPending = abort;
      recognition.onspeechstart = () => { started = performance.now(); };
      recognition.onresult = event => finish({ text: event.results[0][0].transcript, confidence: event.results[0][0].confidence, speechStartedAt: started });
      recognition.onerror = event => finish(undefined, recognitionFailure(event.error));
      recognition.onend = () => finish();
      signal?.addEventListener("abort", abort, { once: true });
      try { recognition.start(); } catch { finish(undefined, "microphone"); }
    }).finally(() => { if (generation === this.generation) this.release(); });
  }
}

/** Match every available choice, never just the correct choice. Recognition is not scoring. */
export type AnswerOption = { id: string; label: string; synonyms?: string[] };
export type AnswerMatch = { kind: "accept" | "confirm"; option: AnswerOption; similarity: number } | { kind: "retry" };
export const SOS_OPTION_ID = "\u0000cogni-sos";

export function optionsWithHelp(options: AnswerOption[], help: string[]): AnswerOption[] {
  return [...options, { id: SOS_OPTION_ID, label: "Help", synonyms: Array.from(new Set(help)) }];
}

export function localizeAnswerOptions(
  options: AnswerOption[],
  localize: (label: string) => string,
  localeSynonyms: Record<string, string[]>,
): AnswerOption[] {
  return options.map(option => {
    const label = localize(option.label);
    return {
      ...option,
      label,
      synonyms: Array.from(new Set([...(option.synonyms ?? []), ...(localeSynonyms[option.label] ?? []), ...(localeSynonyms[label] ?? [])])),
    };
  });
}

export function normalizeAnswer(text: string): string {
  // Preserve vowel marks in Indic scripts. Removing all combining marks corrupts words.
  return text.normalize("NFKC").toLocaleLowerCase().replace(/[\p{P}\p{S}]/gu, " ").replace(/\s+/g, " ").trim();
}

export function similarity(left: string, right: string): number {
  const a = Array.from(normalizeAnswer(left)), b = Array.from(normalizeAnswer(right));
  if (!a.length || !b.length) return 0;
  let previous = Array.from({ length: b.length + 1 }, (_, i) => i);
  a.forEach((letter, i) => {
    const row = [i + 1];
    b.forEach((other, j) => row.push(Math.min(row[j] + 1, previous[j + 1] + 1, previous[j] + (letter === other ? 0 : 1))));
    previous = row;
  });
  return 1 - previous[b.length] / Math.max(a.length, b.length);
}

export function matchAnswer(text: string, options: AnswerOption[], confidence = 0): AnswerMatch {
  const normalized = normalizeAnswer(text);
  if (!normalized) return { kind: "retry" };
  const ranked = options.map(option => ({ option, similarity: Math.max(...[option.label, ...(option.synonyms ?? [])].map(label => similarity(normalized, label))) }))
    .sort((a, b) => b.similarity - a.similarity);
  const best = ranked[0];
  if (!best || best.similarity < 0.65) return { kind: "retry" };
  if (ranked[1] && best.similarity - ranked[1].similarity < 0.12) return { kind: "confirm", ...best };
  // Fuzzy and unknown-confidence transcripts always require confirmation.
  return { kind: best.similarity === 1 && confidence >= 0.8 ? "accept" : "confirm", ...best };
}

export type Locale = {
  code: string; speechLanguage: string; name: string;
  read: string; voice: string; mixed: string; choose: string; mic: string; replay: string;
  listening: string; speaking: string; retry: string; fallback: string; confirm: string;
  yes: string; no: string; heard: string; saved: string; storageError: string;
  help: string[]; synonyms: Record<string, string[]>; ui?: Record<string, string>;
};

/** Translate a visible UI phrase when this locale provides it; English remains the fallback. */
export function translate(locale: Locale, phrase: string, values: Record<string, string | number> = {}): string {
  let translated = locale.ui?.[phrase] ?? phrase;
  const resolvedValues: Record<string, string | number> = { ...values };
  if (!locale.ui?.[phrase]) {
    const templates = Object.entries(locale.ui ?? {}).map(([template, value], index) => ({
      template,
      value,
      index,
      placeholders: Array.from(template.matchAll(/\{([a-zA-Z][a-zA-Z0-9_]*)\}/g)),
      literalLength: template.replace(/\{[a-zA-Z][a-zA-Z0-9_]*\}/g, "").length,
    })).sort((a, b) => b.literalLength - a.literalLength || a.placeholders.length - b.placeholders.length || a.index - b.index);
    for (const { template, value } of templates) {
      const names: string[] = [];
      const pattern = new RegExp(`^${template.split(/\{([a-zA-Z][a-zA-Z0-9_]*)\}/g).map((part, index) => {
        if (index % 2) { names.push(part); return "([\\s\\S]*?)"; }
        return part.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
      }).join("")}$`);
      const match = phrase.match(pattern);
      if (!match) continue;
      translated = value;
      names.forEach((name, index) => {
        if (!Object.prototype.hasOwnProperty.call(values, name)) resolvedValues[name] = match[index + 1];
      });
      break;
    }
  }
  return translated.replace(/\{([a-zA-Z][a-zA-Z0-9_]*)\}/g, (token, name: string) =>
    Object.prototype.hasOwnProperty.call(resolvedValues, name) ? String(resolvedValues[name]) : token,
  );
}

// Server-only discovery: add a JSON file with the same schema, then restart/build.
import { readdirSync, readFileSync } from "node:fs";
import { join } from "node:path";
import type { Locale } from "./locale";
export function loadLocales(): Locale[] {
  const directory = join(process.cwd(), "lib", "interaction", "locales");
  return readdirSync(directory).filter(name => name.endsWith(".json")).sort()
    .map(name => JSON.parse(readFileSync(join(directory, name), "utf8")) as Locale);
}

export type InteractionMode = "read" | "voice" | "mixed";
export type Preferences = { mode: InteractionMode; language: string; textSize: "large" | "larger"; updatedAt: number; override: boolean };
export type KeyValueStore = Pick<Storage, "getItem" | "setItem">;
export const defaultPreferences: Preferences = { mode: "read", language: "en", textSize: "large", updatedAt: 0, override: false };
export const preferenceKey = (patient: string) => `cogni:preferences:${encodeURIComponent(patient)}`;

export function readPreferences(storage: KeyValueStore, patient: string): Preferences {
  try {
    const value = JSON.parse(storage.getItem(preferenceKey(patient)) || "null");
    if (!value || !["read", "voice", "mixed"].includes(value.mode) || !Number.isFinite(value.updatedAt)) return { ...defaultPreferences };
    return { ...defaultPreferences, ...value };
  } catch { return { ...defaultPreferences }; }
}

export function writePreferences(storage: KeyValueStore, patient: string, value: Preferences): void {
  storage.setItem(preferenceKey(patient), JSON.stringify(value));
}

/** Caregiver defaults never erase an explicit patient choice, including mid-game. */
export function mergePreferences(local: Preferences, remote: Preferences): Preferences {
  if (local.override && !remote.override) return local;
  return remote.updatedAt > local.updatedAt ? remote : local;
}

export type InputMethod = "tap" | "typed" | "voice";
export type SessionEvent = {
  id: string; patient: string; session: string; game: string; domain: string;
  timestamp: number; day: string; method: InputMethod; correct: boolean | null;
  responseMs: number | null; hesitationMs?: number | null; touchMs?: number | null;
  errorType: "none" | "answer" | "recognition" | "silence"; difficulty: number;
};
export type SkillState = { rating: number; knob: number; recent: number[]; baselines: Record<InputMethod, number[]>; decision: "stay" | "up" | "down"; reason: string };
export const initialSkill = (): SkillState => ({ rating: 1000, knob: 2, recent: [], baselines: { tap: [], typed: [], voice: [] }, decision: "stay", reason: "Getting to know your pace" });
export function median(values: number[]): number | null {
  if (!values.length) return null;
  const sorted = [...values].sort((a, b) => a - b), mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}
export function adapt(previous: SkillState, event: SessionEvent): SkillState {
  if (event.correct === null || event.errorType === "recognition" || event.errorType === "silence") return previous;
  const state: SkillState = { ...previous, recent: [...previous.recent, Number(event.correct)].slice(-5), baselines: { ...previous.baselines }, decision: "stay", reason: "Keep practising at this setting" };
  const baseline = median(state.baselines[event.method]);
  const expected = 1 / (1 + 10 ** (((800 + 100 * event.difficulty) - state.rating) / 400));
  state.rating = Math.max(400, Math.min(1800, state.rating + 24 * (Number(event.correct) - expected)));
  // A slower response alone can never lower difficulty. Ignore invalid/missing speech-start times.
  const reliableTime = event.responseMs !== null && Number.isFinite(event.responseMs) && event.responseMs >= 0;
  if (reliableTime && event.correct) state.baselines[event.method] = [...state.baselines[event.method], event.responseMs!].slice(-21);
  if (state.recent.length === 5) {
    const accuracy = state.recent.reduce((a, b) => a + b, 0) / 5;
    if (accuracy <= 0.4 && state.knob > 1) {
      state.knob--; state.decision = "down"; state.reason = "More support after several difficult answers"; state.recent = [];
    } else if (accuracy >= 0.8 && state.knob < 4 && (baseline === null || !reliableTime || event.responseMs! <= baseline * 1.5)) {
      state.knob++; state.decision = "up"; state.reason = "Several successful answers at your own pace"; state.recent = [];
    }
  }
  return state;
}
export function slope(values: number[]): number {
  if (values.length < 2) return 0;
  const center = (values.length - 1) / 2, mean = values.reduce((a, b) => a + b, 0) / values.length;
  return values.reduce((sum, value, i) => sum + (i - center) * (value - mean), 0) / values.reduce((sum, _, i) => sum + (i - center) ** 2, 0);
}
/** Require both halves to decline: a single unusually bad final session is insufficient. */
export function declineAlert(scores: number[]): boolean {
  const recent = scores.filter(Number.isFinite).slice(-8);
  if (recent.length < 8) return false;
  return slope(recent) < -2 && slope(recent.slice(0, 4)) < -1 && slope(recent.slice(4)) < -1;
}

# Personal-baseline adaptation

Implementation: `frontend/lib/interaction/adaptive.ts`. It is a pure module so
the same decisions are available offline and directly testable with Node 24.
State is partitioned by patient and domain by `events.ts`. Raw answer events
contain a session ID, game, domain, local day, timestamp, input method,
correctness, response time, error type and the current difficulty knob.

The skill estimate starts at 1000. Expected success is
`1 / (1 + 10 ** ((800 + 100 * difficulty - rating) / 400))`.
Each scored answer updates rating by `24 * (actual - expected)`, clamped to
400–1800. This is an engagement heuristic, not a clinically calibrated model.

A five-answer window controls the knob. Accuracy at or below 40% lowers the
knob once; at or above 80% raises it once if the latest valid response is no
more than 1.5 times that person's baseline. A change clears the window. Knobs
stay between 1 and 4. A slower response alone can never lower difficulty.
The Elo estimate is recorded, while the transparent window policy makes the
actual knob decision; they must not be described as interchangeable.

Baselines are rolling medians of up to 21 correctly answered trials, stored
separately for tap, typed and voice. The comparison uses the prior baseline.
No population speed cutoff is used. Voice time runs from TTS completion to the
browser's `speechstart` event, not transcript delivery. When speech-start timing
is unavailable it remains null. Recognition failures and silence have no
correctness and cannot update skill or baselines. Fuzzy / low-confidence
matches require explicit confirmation before invoking the game's normal
answer handler. Missing timing does not penalize the patient.

Odd One Out and Object Naming apply their knobs to choice count at the next
question. Matching Pictures chooses pair count at session start; Sound Tap
adjusts beat range, and Pattern Completion switches between two- and three-item
patterns. Yesterday Quiz adds up to three older journal memories as distractors,
excluding every target-day memory and normalized summaries repeated on that day.
When there is insufficient safe history it uses fewer choices rather than inventing
memories. Its answer positions still rotate. Crossword uses word length and given
letters at puzzle creation. Catalog levels remain unchanged. Other instrumented
legacy games log events but retain their existing mechanics.
The event schema reserves hesitation and

import { normalizeAnswer } from "./matching.ts";
export type YesterdayMemory = { id: string; label: string; day: string };

export function validMemoryDay(day: unknown): day is string {
  if (typeof day !== "string" || !/^\d{4}-\d{2}-\d{2}$/.test(day)) return false;
  const [year, month, date] = day.split("-").map(Number);
  const local = new Date(year, month - 1, date, 12);
  return local.getFullYear() === year && local.getMonth() === month - 1 && local.getDate() === date;
}

const noRecallOption = {
  id: "dont-remember",
  label: "I don't remember",
  synonyms: ["not sure", "I cannot remember"],
};

export function buildYesterdayChoices(
  memories: YesterdayMemory[],
  answerId: string,
  knob: number,
  roundIndex: number,
) {
  const answer = memories.find(memory => memory.id === answerId);
  if (!answer || answerId === noRecallOption.id || !validMemoryDay(answer.day) || !normalizeAnswer(answer.label)) {
    throw new Error("Yesterday Quiz answer must match a saved memory");
  }

  const level = Math.max(1, Math.min(4, Math.trunc(Number.isFinite(knob) ? knob : 1)));
  const seen = new Set([answerId, noRecallOption.id]);
  // Every memory from the target day answers the question truthfully. Also
  // exclude earlier copies of those summaries, so repeated routines are fair.
  const seenLabels = new Set(memories.filter(memory => memory.day === answer.day).map(memory => normalizeAnswer(memory.label)));
  const distractors: YesterdayMemory[] = [];
  for (const memory of memories) {
    if (distractors.length >= level - 1) break;
    const label = normalizeAnswer(memory.label);
    if (seen.has(memory.id) || !label || seenLabels.has(label) || !validMemoryDay(memory.day) || memory.day >= answer.day) continue;
    seen.add(memory.id);
    seenLabels.add(label);
    distractors.push(memory);
  }

  const choices = [answer, ...distractors, noRecallOption];
  const offset = ((roundIndex + 1) % choices.length + choices.length) % choices.length;
  return [...choices.slice(offset), ...choices.slice(0, offset)];
}

@api.route('/accuracy')
def accuracy():
    given = request.args.get('given')
    real = request.args.get('real')
    if not given or not real:
        return jsonify(error="Both answers are required"), 400
    video_path = request.args.get('video')  # Get video path to find journal
    journal = None
    
    # Try to get journal entry for additional context
    if video_path:
        try:
            # Extract video name from path
            video_name = video_path.split('/')[-2].replace('_chunks', '')
            journal_file = 'video_journals.json'
            if os.path.exists(journal_file):
                with open(journal_file, 'r') as f:
                    journals = json.load(f)
                    # Try to find matching journal
                    for vid_name, journal_data in journals.items():
                        if video_name in vid_name or vid_name in video_name:
                            journal = journal_data.get('entry')
                            break
        except:
            pass  # If journal retrieval fails, continue without it
    
    score = gpt_accuracy(given, real, journal)
    # Ensure score is an integer
    try:
        score_int = int(score)
        if not 0 <= score_int <= 10:
            raise ValueError("Score out of range")
    except (ValueError, TypeError):
        return jsonify(error="Answer evaluation is unavailable. Your answer has not been scored."), 503
    
    return jsonify(score_int)


def gpt_accuracy(answer1, answer2, journal_context=None):
    # An unavailable evaluator must never fabricate cognitive performance.
    if not client.api_key or client.api_key == 'not_configured':
        return None
    try:
        # Build the prompt with journal context if available
        prompt = (
            f"Determine the accuracy of the given answer compared to the real answer. "
            "Accuracy should be on a scale of 1 to 10 with 10 being almost the same. "
            "Allow for slight variation and different phrasing. "
            "Award more points for capturing the correct main idea and key details. "
        )
        
        if journal_context:
            prompt += f"\n\nAdditional context from journal entry: {journal_context}\n"
        
        prompt += f"\nGiven answer: {answer1}\nTrue answer: {answer2}\n\nOnly return the number (1-10)."
        
        response = client.chat.completions.create(
            model='gpt-4o-mini',
            messages=[
                {"role": "system", "content": "You are an accuracy evaluator for memory quiz answers. Consider the semantic meaning and key details rather than exact wording."},
                {"role": "user", "content": prompt}
            ]
        )
        return response.choices[0].message.content
    except Exception as e:
        return f"Error with OpenAI API: {e}"

import unittest

import app


class AccuracyAvailabilityTests(unittest.TestCase):
    def test_missing_key_never_fabricates_a_score(self):
        with patch.object(app.client, 'api_key', 'not_configured'):
            result = app.api.test_client().get('/accuracy?given=tea&real=tea')
        self.assertEqual(result.status_code, 503)
        self.assertIn('not been scored', result.get_json()['error'])

    def test_evaluator_error_and_invalid_score_are_not_cognitive_errors(self):
        for value in ('Error: offline', '100', None):
            with patch.object(app, 'gpt_accuracy', return_value=value):
                self.assertEqual(app.api.test_client().get('/accuracy?given=tea&real=tea').status_code, 503)

    def test_valid_evaluator_score_is_preserved(self):
        with patch.object(app, 'gpt_accuracy', return_value='8'):
            result = app.api.test_client().get('/accuracy?given=tea&real=tea')
        self.assertEqual(result.get_json(), 8)
        self.assertEqual(app.api.test_client().get('/accuracy').status_code, 400)

import { API_URL } from "@/lib/api";
export async function POST(request: Request) {
  try {
    const req = await request.json();

    const value = req.given;
    const answer = req.real;
    const video = req.video || '';

    const params = new URLSearchParams({
      given: value,
      real: answer + '.',
      ...(video && { video })
    });

    const res = await fetch(
      `${API_URL}/accuracy?${params.toString()}`,
      {
        headers: {
          "Content-Type": "application/json",
        },
        cache: "no-store",
      },
    );
    const data = await res.json();
    if (!res.ok) return Response.json({ error: data.error || "Evaluation unavailable" }, { status: res.status });

    return Response.json({ data });
  } catch (err) {
    console.log(err);
    return Response.json({ error: "Failed to check accuracy" }, { status: 500 });
  }
}
